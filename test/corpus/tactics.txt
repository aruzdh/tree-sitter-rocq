Theorem eqb_list_true_iff :
  forall A (eqb : A -> A -> bool),
    (forall a1 a2, eqb a1 a2 = true <-> a1 = a2) ->
    forall l1 l2, eqb_list eqb l1 l2 = true <-> l1 = l2.

  intros A P Heqb. split.
  - generalize l2. induction l1 as [| x l' IHl'].
    + intros l H. destruct l as [| x l'].
      * reflexivity.
      * discriminate.
    + intros l H. destruct l as [| x' l''].
      * discriminate.
      * f_equal.
        -- apply Heqb. simpl in H. 
          apply andb_true_iff in H as [H _]. apply H.
        -- apply IHl'. simpl in H.
            apply andb_true_iff in H as [_ H]. apply H.
  - generalize l2. induction l1 as [| x l' IHl'].
    + intros l H. rewrite <- H. reflexivity.
    + intros l H. destruct l as [| x' l''].
      * discriminate.
      * simpl. apply andb_true_iff. injection H as Hh Ht. split.
        -- apply Heqb. apply Hh.
        -- apply IHl'. apply Ht.
Qed.

Theorem optimize_0times_sound : forall a,
  aeval (optimize_0times a) = aeval a.
Proof.
  intro a. 
  induction a; 
  try (simpl; rewrite IHa1, IHa2; reflexivity);
  try reflexivity.
  (* AMult case *)
  - destruct a1 eqn:Ea1; 
    try (simpl; simpl in IHa1; rewrite IHa1, IHa2;
        reflexivity).
    (* AMult case *)
    + destruct n eqn:En;
      [reflexivity | simpl; rewrite IHa2; reflexivity].
Qed.
